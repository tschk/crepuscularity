
    private fun videoValue(method: String): JSONObject {
        if (method != "capture") error("unsupported video method: $method")
        pendingVideoAction = "video.capture"
        captureVideo?.invoke() ?: emit(errorJson("video.capture", "video capture unavailable"))
        return JSONObject().put("opening", true)
    }

    private fun videoResultJson(action: String, uri: Uri): String {
        val name = queryDisplayName(uri)
        val mime = appContext.contentResolver.getType(uri) ?: "video/mp4"
        val file = copyToCache(uri, name, mime)
        return JSONObject().put("ok", true).put("action", action).put("value", JSONObject().put("files", JSONArray(listOf(JSONObject()
            .put("name", file.name).put("mimeType", mime).put("bytes", file.length())
            .put("filePath", file.absolutePath).put("importSource", "android-video-camera").put("mediaKind", "video"))))).toString()
    }
